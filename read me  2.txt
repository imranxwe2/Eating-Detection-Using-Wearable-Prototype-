code to convert avi videos into imaegs

Get-ChildItem -Filter "*.avi" | ForEach-Object {
    $name = $_.BaseName
    $folder = Join-Path $_.DirectoryName $name

    if (-not (Test-Path $folder)) {
        New-Item -ItemType Directory -Path $folder | Out-Null
    }

    Write-Host "Processing: $($_.Name)" -ForegroundColor Cyan

    ffmpeg -i $_.FullName -vf "fps=2" (Join-Path $folder "frame_%06d.jpg")
}
